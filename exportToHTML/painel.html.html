    document.addEventListener("DOMContentLoaded", function() {
        var qrElements = document.querySelectorAll('.qrcode');

        qrElements.forEach(function(el) {
            var texto = el.getAttribute('data-text');
            if (texto) {
                new QRCode(el, {
                    text: texto,
                    width: 100,
                    height: 100,
                    colorDark : "#000000",
                    colorLight : "#ffffff",
                    correctLevel : QRCode.CorrectLevel.H
                });
            }
        });
    });
</script>
